CREATE OR REPLACE DATABASE RETAIL_DB;

CREATE OR REPLACE SCHEMA RETAIL_DB.STAGING_LAYER;
CREATE OR REPLACE SCHEMA RETAIL_DB.CLEANSED_LAYER;
CREATE OR REPLACE SCHEMA RETAIL_DB.BUSINNESS_LAYER;
CREATE OR REPLACE SCHEMA RETAIL_DB.PRESENTATION_LAYER;

CREATE FILE FORMAT STAGING_LAYER.CSV_FORMAT;
    TYPE = "CSV"
    DELIMITER = ','
    SKIP_HEADER_ROWS = 1
   );

CREATE STAGE.RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE;
    URL: 'S3://etl-and-data-warehouse/'
    CREDENTIALS = (
        AWS_KEY_ID = ''
        AWS_SECRET_ACCESS_KEY = ''
    );
    FILE_FORMAT = RETAIL_DB.STAGING_LAYER.CSV_FORMAT;

CREATE OR REPLACE TABLE RETAIL_DB.CLEANSED_LAYER.CLEANED_SALES(
    SALES_ID INT,
    PRODUCT_ID INT,
    REGION STRING,
    PRICE NUMERIC(10,2),
    QUANTITY INT,
    TIMESTAMP DATE,
    TOTAL_SALES NUMERIC(10,2)
);

CREATE OR REPLACE TABLE RETAIL_DB.CLEANSED_LAYER.CLEANED_PRODUCTS(
    PRODUCT_ID INT,
    BRAND STRING,
    CATEGORY STRING,
    RATING NUMERIC(10,2)
);

CREATE OR REPLACE TABLE RETAIL_DB.BUSINESS_LAYER.ENRICHED_SALES(
    SALES_ID INT,
    PRODUCT_ID INT,
    REGION STRING,
    PRICE NUMERIC(10,2),
    QUANTITY INT,
    TIMESTAMP DATE,
    TOTAL_SALES NUMERIC(10,2),
    BRAND STRING,
    CATEGORY STRING,
    RATING NUMERIC(10,2),
    MONTH STRING,
    WEEK INT,
    WEEKDAY STRING,
    HOUR INT,
    SALES_BUCKET STRING
);

CREATE OR REPLACE TABLE RETAIL_DB.PRESENTATION_LAYER.HOURLY_SALES_TREND(
    REGION STRING,
    CATEGORY STRING,
    HOUR NUMBER(2, 0),
    HOURLY_TOTAL_SALES NUMERIC(10, 2),
    SALES_BUCKET STRING,
);

CREATE OR RPEPLACE TABLE RETAIL_DB.PRESENTATION_LAYER.SEASONAL_SALES_PATTERN(
    QUARTER INT,
    CATEGORY STRING,
    TOTAL_SALES NUMERIC(10, 2)
);
CREATE OR REPLACE TABLE RETAIL_DB.PRESENTATION_LAYER.REVENUE_CONCENTRATION(
    REGION STRING,
    REGION_REVENUE NUMERIC(10, 2),
    REVENUE_SHARE NUMERIC(10, 2),
    CUMULATIVE_REVENUE_SHARE NUMERIC(10, 2)
);

CREATE OR REPLACE TABLE RETAIL_DB.PRESENTATION_LAYER.SALES_RANKING(
    PRODUCT_ID INT,
    REVENUE NUMERIC(10, 2),
    SALES_COUNT INT,
    VALUE_BUCKET STRING
);

COPY INTO RETAIL_DB.CLEANSED_LAYER.CLEANED_SALES
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Outputs/
FILES = ('cleaned_sales.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.CLEANED_PRODUCTS
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Outputs/
FILES = ('cleaned_products.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.ENRICHED_SALES
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Outputs/
FILES = ('enrich.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.HOURLY_SALES_TREND
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Analytics/
FILES = ('hourly_sales_trend.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.PRODUCT_SALES_RANKING
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Analytics/
FILES = ('product_sales_ranking.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.REVENUE_CONCENTRATION
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Analytics/
FILES = ('revenue_concentration.csv'),
ON ERROR = CONTINUE;

COPY INTO RETAIL_DB.CLEANSED_LAYER.SEASONAL_SALES_PATTERNS
FROM @RETAIL_DB.STAGING_LAYER.RETAIL_S3_STAGE/Analytics/
FILES = ('seasonal_sales_patterns.csv'),
ON ERROR = CONTINUE;
